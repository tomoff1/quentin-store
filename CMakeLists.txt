cmake_minimum_required(VERSION 3.20)
project(QuentinStore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
add_executable(core-tests tests/core_tests.cpp)
target_include_directories(core-tests PRIVATE src third_party)
add_test(NAME core-security COMMAND core-tests)
option(QS_DESKTOP "Build desktop SDL client for integration testing" OFF)
if(QS_DESKTOP)
  find_package(SDL2 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(QS_CURL REQUIRED IMPORTED_TARGET libcurl)
  find_package(OpenSSL REQUIRED)
  find_package(Threads REQUIRED)
  find_library(SDL2_GFX_LIBRARY SDL2_gfx REQUIRED)
  find_library(SDL2_TTF_LIBRARY SDL2_ttf REQUIRED)
  add_executable(quentin-store src/main.cpp src/network.cpp)
  target_include_directories(quentin-store PRIVATE src third_party ${SDL2_INCLUDE_DIRS})
  target_link_libraries(quentin-store PRIVATE SDL2::SDL2 ${SDL2_GFX_LIBRARY} ${SDL2_TTF_LIBRARY} PkgConfig::QS_CURL OpenSSL::Crypto Threads::Threads)
  add_executable(network-tests tests/network_tests.cpp src/network.cpp)
  target_include_directories(network-tests PRIVATE src third_party)
  target_link_libraries(network-tests PRIVATE PkgConfig::QS_CURL OpenSSL::Crypto Threads::Threads)
  add_executable(public-catalog-test tests/public_catalog_test.cpp src/network.cpp)
  target_include_directories(public-catalog-test PRIVATE src third_party)
  target_link_libraries(public-catalog-test PRIVATE PkgConfig::QS_CURL OpenSSL::Crypto Threads::Threads)
endif()
